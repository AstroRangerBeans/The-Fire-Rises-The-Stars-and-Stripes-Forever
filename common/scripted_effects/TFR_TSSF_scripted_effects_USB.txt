USB_add_corporate_influence = {
	if = {
		limit = {
			NOT = {
				has_country_flag = USB_corporate_influence_disabled_flag
			}
		}
		custom_effect_tooltip = USB_add_corporate_influence_tooltip
		add_to_variable = {
			var = USB_corporate_influence_var
			value = USB_corporate_influence_var_temp
		}
		clamp_variable = {
			var = USB_corporate_influence_var
			min = 0
			max = 1
		}
		hidden_effect = {
			remove_ideas = USB_lower_corporate_influence_idea
			remove_ideas = USB_low_corporate_influence_idea
			remove_ideas = USB_medium_corporate_influence_idea
			remove_ideas = USB_high_corporate_influence_idea
			remove_ideas = USB_higher_corporate_influence_idea
			if = {
				limit = {
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.2
						compare = less_than
					}
				}
				add_ideas = USB_lower_corporate_influence_idea
			}
			else_if = {
				limit = {
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.2
						compare = greater_than_or_equals
					}
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.4
						compare = less_than
					}
				}
				add_ideas = USB_low_corporate_influence_idea
			}
			else_if = {
				limit = {
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.4
						compare = greater_than_or_equals
					}
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.6
						compare = less_than
					}
				}
				add_ideas = USB_medium_corporate_influence_idea
			}
			else_if = {
				limit = {
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.6
						compare = greater_than_or_equals
					}
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.8
						compare = less_than
					}
				}
				add_ideas = USB_high_corporate_influence_idea
			}
			else_if = {
				limit = {
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.8
						compare = greater_than_or_equals
					}
					check_variable = {
						var = USB_corporate_influence_var
						value = 0.95
						compare = less_than
					}
				}
				add_ideas = USB_higher_corporate_influence_idea
			}
			else_if = {
				limit = {
					NOT = {
						has_country_flag = USB_reunified_flag

	clear_variable = election_electoral_votes_d
	clear_variable = election_electoral_votes_p
	clear_variable = election_electoral_votes_i
	set_variable = { electoral_var_race_r = 538 }
	clear_variable = electoral_var_race_d
	clear_variable = electoral_var_race_p
	clear_variable = electoral_var_race_i
	
	while_loop_effect = {
		limit = {
			check_variable = { current_state < 52 }	# 50 states + DC
		}
		
		set_variable = { usa_election_state_winner^current_state = current_state }
		
		if = {
			limit = {
				check_variable = { usa_election_state_major^current_state = 2 }
			}
			add_to_variable = { election_electoral_votes_d = usa_election_state_vote^current_state }
	
			add_to_variable = { electoral_var_race_d = usa_election_state_vote^current_state }
			add_to_variable = { electoral_var_race_p = usa_election_state_vote^current_state }
			add_to_variable = { electoral_var_race_i = usa_election_state_vote^current_state }
		}
		else_if = {
			limit = {
				check_variable = { usa_election_state_major^current_state = 3 }
			}
			add_to_variable = { election_electoral_votes_r = usa_election_state_vote^current_state }
			
			subtract_from_variable = { electoral_var_race_r = usa_election_state_vote^current_state }
		}
		else = {
			add_to_variable = { election_electoral_votes_p = usa_election_state_vote^current_state }
			add_to_variable = { electoral_var_race_p = usa_election_state_vote^current_state }
		}
		else = {
			add_to_variable = { election_electoral_votes_i = usa_election_state_vote^current_state }
			add_to_variable = { electoral_var_race_i = usa_election_state_vote^current_state }
		}
		
		add_to_temp_variable = { current_state = 1 }
	}

# Update the map according to who has the majority popularity each state
USA_set_majority_according_to_local = {
	for_loop_effect = {
		start = 1
		end = 52
		value = current_state
		
		if = {
			limit = {
				check_variable = { usa_election_popularity_d^current_state > usa_election_popularity_r^current_state }
				check_variable = { usa_election_popularity_d^current_state > usa_election_popularity_f^current_state }
			}
			set_variable = { usa_election_state_major^current_state = 1 }
		}
		else_if = {
			limit = {
				check_variable = { usa_election_popularity_r^current_state > usa_election_popularity_d^current_state }
				check_variable = { usa_election_popularity_r^current_state > usa_election_popularity_p^current_state }
				check_variable = { usa_election_popularity_r^current_state > usa_election_popularity_i^current_state }
			}
			set_variable = { usa_election_state_major^current_state = 2 }
		}
		else = {
			set_variable = { usa_election_state_major^current_state = 4 }
		}
	}
}

		set_variable = { usa_election_popularity_f^current_state = temp_popularity_total }

USA_election_projection_get_ceiling_floor = {
	set_temp_variable = { projection_y_ceiling = 0 }
	set_temp_variable = { projection_y_floor = projection_y_position_d_list^0 }
	
	for_each_loop = {

		value = v
		
		if = {
			limit = { check_variable = { v > projection_y_ceiling } }
			
			set_temp_variable = { projection_y_ceiling = v }
		}
		else_if = {
			limit = { check_variable = { v < projection_y_floor } }
			
			set_temp_variable = { projection_y_floor = v }
		}
	}
	for_each_loop = {
		array = projection_y_position_r_list
		value = v
		
		if = {

			set_temp_variable = { projection_y_ceiling = v }
		}
		else_if = {
			limit = { check_variable = { v < projection_y_floor } }
			
			set_temp_variable = { projection_y_floor = v }
		}
	}
	for_each_loop = {
		array = projection_y_position_p_list
		value = v
		
		if = {
			limit = { check_variable = { v > projection_y_ceiling } }
			
			set_temp_variable = { projection_y_ceiling = v }
		}
		else_if = {
			limit = { check_variable = { v < projection_y_floor } }
			
			set_temp_variable = { projection_y_floor = v }

			set_temp_variable = { projection_y_ceiling = v }
		}
		else_if = {
			limit = { check_variable = { v < projection_y_floor } }
			
			set_temp_variable = { projection_y_floor = v }
